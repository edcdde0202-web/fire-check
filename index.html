    viewport.onwheel = function(e) {

        e.preventDefault();

        if (e.ctrlKey) { mapX += (e.deltaY < 0 ? 55 : -55); updateTransform(); return; }

        const rect = viewport.getBoundingClientRect();

        const mouseX = e.clientX - rect.left, mouseY = e.clientY - rect.top;

        const currentX = (mouseX - mapX) / mapScale, currentY = (mouseY - mapY) / mapScale;

        mapScale = Math.min(Math.max(0.1, mapScale * (e.deltaY < 0 ? 1.1 : 1/1.1)), 10); 

        mapX = mouseX - currentX * mapScale; mapY = mouseY - currentY * mapScale;

        updateTransform();

    };

    function calculateExpiryStatus(dateString, yearsLimit) {

        if (!dateString) return { status: 'normal' };

        const today = new Date(); today.setHours(0,0,0,0);

        const mfgDate = new Date(dateString); mfgDate.setHours(0,0,0,0);

        const expiryDate = new Date(mfgDate); expiryDate.setFullYear(expiryDate.getFullYear() + yearsLimit);

        if (today >= expiryDate) return { status: 'expired' };

        return { status: 'normal' };

    }
